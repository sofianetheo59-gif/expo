import React, { useState } from 'react';
import { View, Text, Image, PanResponder, Animated, StyleSheet, TouchableOpacity } from 'react-native';

export default function App() {
  const [elements, setElements] = useState([
    {
      id: 1,
      type: 'image',
      uri: 'https://images.unsplash.com/photo-1593642532973-d31b6557fa68?auto=format&fit=crop&w=100&q=80',
      position: new Animated.ValueXY({ x: 50, y: 50 }),
    },
    {
      id: 2,
      type: 'text',
      text: 'Mon texte',
      position: new Animated.ValueXY({ x: 150, y: 200 }),
    },
  ]);

  const createPanResponder = (element) =>
    PanResponder.create({
      onStartShouldSetPanResponder: () => true,
      onPanResponderMove: Animated.event(
        [null, { dx: element.position.x, dy: element.position.y }],
        { useNativeDriver: false }
      ),
      onPanResponderRelease: () => {},
    });

  return (
    <View style={styles.container}>
      {elements.map((el) => (
        <Animated.View
          key={el.id}
          style={[el.position.getLayout(), styles.box]}
          {...createPanResponder(el).panHandlers}
        >
          {el.type === 'image' && <Image source={{ uri: el.uri }} style={{ width: 100, height: 100 }} />}
          {el.type === 'text' && <Text style={styles.text}>{el.text}</Text>}
        </Animated.View>
      ))}

      <TouchableOpacity
        style={styles.button}
        onPress={() =>
          setElements([
            ...elements,
            {
              id: elements.length + 1,
              type: 'text',
              text: 'Nouveau texte',
              position: new Animated.ValueXY({ x: 100, y: 100 }),
            },
          ])
        }
      >
        <Text style={{ color: 'white' }}>Ajouter Texte</Text>
      </TouchableOpacity>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, backgroundColor: '#fff' },
  box: { position: 'absolute' },
  text: { color: 'black', fontSize: 18 },
  button: {
    position: 'absolute',
    bottom: 40,
    left: 20,
    backgroundColor: 'blue',
    padding: 15,
    borderRadius: 10,
  },
});
